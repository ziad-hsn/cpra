package legacy_test

import (
	"context"
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"

	"github.com/ziad-hsn/cpra/sdk/go/legacy"
)

// This example uses a local HTTP fixture so it runs without a CPRa process.
// Configure the actual server address and token to read a running v1 server.
func ExampleClient_ListMonitors() {
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodGet || r.URL.Path != "/api/v1/monitors" ||
			r.Header.Get("Authorization") != "Bearer fixture-token" {
			http.Error(w, "unexpected request", http.StatusBadRequest)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		io.WriteString(w, `{"monitors":[],"page":1,"size":100,"total":0}`)
	}))
	defer server.Close()

	client, err := legacy.New(legacy.Config{
		BaseURL: server.URL, AuthToken: "fixture-token",
	})
	if err != nil {
		panic(err)
	}
	page, err := client.ListMonitors(context.Background(), legacy.MonitorListOptions{
		Page: 1, Size: 100,
	})
	if err != nil {
		panic(err)
	}
	fmt.Println("monitors:", len(page.Monitors))
	// Output: monitors: 0
}
